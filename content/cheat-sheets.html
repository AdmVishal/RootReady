Files & filesystems

# usage and type
df -hT
df -i                                   # inodes
du -xh --max-depth=1 /var | sort -h     # biggest directories, one filesystem
find / -xdev -type f -size +500M        # big files
lsof +L1                                # deleted files still open
findmnt                                 # mount tree
findmnt --verify                        # validate /etc/fstab
mount -o remount,rw /mountpoint         # remount read-write
# permissions
ls -lZ file                             # mode, owner and SELinux context
chmod 640 file; chmod -R g+rX dir
chown user:group file
setfacl -m u:alice:rwx dir; getfacl dir
# archives and copy
tar -czf out.tgz dir/; tar -xzf out.tgz -C /dest; tar -tzf out.tgz | head
rsync -avh --progress src/ dst/         # add -n for a dry run

Disks & partitions

# inspect (safe)
lsblk -f
blkid
fdisk -l
parted -l
smartctl -H /dev/sdX                     # smartmontools
# make a filesystem (destructive: check the device twice)
mkfs.xfs /dev/sdb1
mkfs.ext4 /dev/sdb1
# mount permanently: add a UUID line to /etc/fstab, then test
mount -a
# grow a filesystem
xfs_growfs /mountpoint                   # XFS (mounted)
resize2fs /dev/mapper/vg-lv              # ext4
# repair (unmounted only)
xfs_repair -n /dev/sdb1                  # dry run first
fsck -f /dev/sdb1
# rescan for new SCSI/SAN disks
echo "- - -" | tee /sys/class/scsi_host/host*/scan

LVM & multipath

# overview
pvs; vgs; lvs
lvs -o +devices
# create
pvcreate /dev/sdb
vgcreate vg01 /dev/sdb
lvcreate -n lv_data -L 20G vg01
mkfs.xfs /dev/vg01/lv_data
# extend (online); -r also grows the filesystem
vgextend vg01 /dev/sdc
lvextend -r -L +5G /dev/vg01/lv_data
lvextend -r -l +100%FREE /dev/vg01/lv_data
# snapshot
lvcreate -s -n snap1 -L 2G /dev/vg01/lv_data
# remove (destructive)
lvremove /dev/vg01/snap1
# multipath
multipath -ll
multipathd show paths
multipath -r                              # reload maps
